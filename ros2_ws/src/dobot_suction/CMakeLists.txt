cmake_minimum_required(VERSION 3.16)
project(dobot_suction)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
add_library(dobot_suction SHARED src/suction_system.cpp)
target_compile_features(dobot_suction PRIVATE cxx_std_17)
target_link_libraries(dobot_suction PRIVATE gz-sim8::gz-sim8
  gz-plugin2::register gz-transport13::gz-transport13)
install(TARGETS dobot_suction LIBRARY DESTINATION lib)
ament_package()
